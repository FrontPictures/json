option(JSON_Valgrind "Execute test suite with Valgrind." OFF)
option(JSON_FastTests "Skip expensive tests." OFF)
include(download_test_data)
add_test(NAME download_test_data COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target download_test_data)
set_tests_properties(download_test_data PROPERTIES FIXTURES_SETUP TEST_DATA)

add_library(test_main OBJECT src/unit.cpp)
target_compile_definitions(test_main PUBLIC DOCTEST_CONFIG_SUPER_FAST_ASSERTS)
target_include_directories(test_main PUBLIC thirdparty/doctest thirdparty/fifo_map ${PROJECT_BINARY_DIR}/include)
target_link_libraries(test_main PUBLIC nlohmann_json)
if(MSVC)
    target_compile_options(test_main PUBLIC /EHsc /W4 /wd4566 /wd4996 /bigobj)
endif()
if(JSON_Valgrind)
    find_program(JSON_VALGRIND_EXECUTABLE valgrind REQUIRED)
endif()

file(GLOB files CONFIGURE_DEPENDS src/unit-*.cpp)
foreach(file IN LISTS files)
    get_filename_component(name ${file} NAME_WE)
    string(REPLACE "unit-" "test-" name ${name})
    set(target ${name}_cpp20)
    add_executable(${target} ${file})
    target_link_libraries(${target} PRIVATE test_main)
    set(args)
    if(NOT JSON_FastTests)
        list(APPEND args --no-skip)
    endif()
    if(JSON_Valgrind)
        add_test(NAME ${target} COMMAND ${JSON_VALGRIND_EXECUTABLE} --error-exitcode=1 --leak-check=full $<TARGET_FILE:${target}> ${args})
    else()
        add_test(NAME ${target} COMMAND ${target} ${args})
    endif()
    set_tests_properties(${target} PROPERTIES WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} FIXTURES_REQUIRED TEST_DATA TIMEOUT 180)
    if(MSVC AND name MATCHES "test-(cbor|msgpack|ubjson|bjdata|binary_formats)")
        target_link_options(${target} PRIVATE /STACK:4000000)
    endif()
endforeach()
set_tests_properties(test-unicode4_cpp20 PROPERTIES TIMEOUT 3000)
